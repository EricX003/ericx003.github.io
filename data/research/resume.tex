\newcommand{\pub}[3][]{%
  \item%

  #2%
  \if\relax\detokenize{#3}\relax

% \pubcite{key1,key2,...} -> "[1, 3, 7]".  Indices come from the \pub keys and
% are sorted numerically, so the list stays in order when entries are added or
% move between the Under Review and Publications lists.  Undefined keys warn in
% the log and render as 0 (they also read as 0 on the first of the two passes).
\ExplSyntaxOn
\msg_new:nnn { xing } { pubcite-undefined }
  { Publication~key~'#1'~is~undefined~--~check~the~\string\pub\space key. }
\clist_new:N \l__xing_pubcite_clist
% enumitem's ref= stores the value brace-wrapped ({7}), so strip one brace level
% with \tl_head:f before doing arithmetic on it.
\cs_new:Npn \__xing_pubnum:n #1
  { \int_eval:n { \tl_head:f { \getrefnumber { pub:#1 } } } }
\cs_new_protected:Npn \__xing_pubcite_add:n #1
  {
    \refused { pub:#1 }   % make a stale/cold .aux trigger LaTeX's rerun warning
    \int_compare:nNnT { \__xing_pubnum:n {#1} } = { 0 }
      { \msg_warning:nnn { xing } { pubcite-undefined } {#1} }
    \clist_put_right:Nx \l__xing_pubcite_clist { \__xing_pubnum:n {#1} }
  }
\NewDocumentCommand \pubcite { m }
  {
    \clist_clear:N \l__xing_pubcite_clist
    \clist_map_inline:nn {#1} { \__xing_pubcite_add:n {##1} }
    \clist_sort:Nn \l__xing_pubcite_clist
      {
        \int_compare:nNnTF {##1} > {##2}
          { \sort_return_swapped: }
          { \sort_return_same: }
      }
    \nobreakspace [ \clist_use:Nn \l__xing_pubcite_clist { ,~ } ]
  }
\ExplSyntaxOff

% The publist environment is one continuously numbered list split across two
% sections (Selected Publications, Under Review).  The first instance in the
% document starts the series and every later one resumes it, so the sections can
% be reordered freely.  Hard-coding series= on one list and resume*= on the other
% breaks the build the moment the two sections swap places.
\newif\ifxingpubliststarted
\newenvironment{publist}
  {%
    \ifxingpubliststarted
      \enumerate[resume*=publist]%
    \else
      \global\xingpubliststartedtrue
      \enumerate[series=publist,label={[\arabic*]}\parindent-0.2\labelwidth\relax,ref=\arabic*]%
    \fi
  }
  {\endenumerate}
